(function () {
  const SUPABASE_URL = "https://pgpweaerodprcsgfztzl.supabase.co";
  const SUPABASE_KEY = "sb_publishable_kpZLHzPc-NcQ8g9A3Cl9JQ_JzCjZb4h";
  const EXCLUDE_KEY = "yashfeen_exclude_device";
  const VISITOR_KEY = "yashfeen_visitor_id";

  if (localStorage.getItem(EXCLUDE_KEY) === "1") return;

  function uuid() {
    if (typeof crypto !== "undefined" && crypto.randomUUID) return crypto.randomUUID();
    return 'xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx'.replace(/[xy]/g, function(c) {
      const r = Math.random() * 16 | 0;
      const v = c === 'x' ? r : (r & 0x3 | 0x8);
      return v.toString(16);
    });
  }

  let visitorId = localStorage.getItem(VISITOR_KEY);
  if (!visitorId) {
    visitorId = uuid();
    localStorage.setItem(VISITOR_KEY, visitorId);
  }

  fetch(SUPABASE_URL + "/rest/v1/site_visits", {
    method: "POST",
    headers: {
      "apikey": SUPABASE_KEY,
      "Authorization": "Bearer " + SUPABASE_KEY,
      "Content-Type": "application/json",
      "Prefer": "return=minimal"
    },
    body: JSON.stringify({ visitor_id: visitorId })
  }).catch(function() {});
